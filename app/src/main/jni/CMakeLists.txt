cmake_minimum_required(VERSION 3.22.1)
project(imdetect)
set(CMAKE_CXX_STANDARD 17)
add_library(imdetect SHARED imdetect.cpp)
find_library(log-lib log)

set(OpenCV_DIR "${CMAKE_SOURCE_DIR}/../../../../opencv/OpenCV-android-sdk/sdk/native/jni")
if(EXISTS ${OpenCV_DIR})
    include(${OpenCV_DIR}/OpenCVConfig.cmake)
    target_include_directories(imdetect PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(imdetect ${OpenCV_LIBS} ${log-lib})
else()
    message(FATAL_ERROR "OpenCV SDK not found")
endif()
